def step_1(input_list):
    """
    Filters log entries for events related to the at.exe process from the specified index.
    
    Parameters:
    input_list (list of dict): List of log entries to process
    
    Returns:
    list of dict: Filtered log entries matching the criteria
    """
    # Define the index value to filter on
    target_index = "__your_sysmon_index__"
    
    # Define the pattern for the Image field
    target_image = "C:\\Windows\\*\\at.exe"
    
    # Initialize a list to store filtered entries
    filtered_entries = []
    
    # Process each log entry in the input list
    for entry in input_list:
        # Check if the index matches the target index
        if entry.get("index") == target_index:
            # Check if the Image field matches the pattern
            if entry.get("Image") == target_image:
                # If both conditions are met, add the entry to the filtered list
                filtered_entries.append(entry)
    
    # Return the filtered list of entries
    return filtered_entries
